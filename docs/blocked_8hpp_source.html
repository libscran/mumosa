53/*
54 * We don't apply block-specific scaling factors as we don't want to alter the relative values within the same modality.
55 * We shouldn't have to do it in the first place - as it's the same modality! - but more importantly, we could introduce spurious differences between blocks.
56 * In the simplest case, two blocks have the same subpopulation structure but the number of cells is different.
57 * We would get different distances in each block due to density, causing us to scale each block differently.
58 * More generally, we could expect differences in subpopulation structure between blocks, leading to different distances even in the absence of any batch effects.
59 * (Mind you, differences in subpopulation structure also interfere with accurate scaling between modalities,
60 * but any errors in scaling modalities are much less obvious than those from scaling blocks.)
61 */

86template<typename Index_, typename Distance_>
87std::pair<Distance_, Distance_> compute_distance_blocked(const std::vector<std::pair<Index_, Distance_*> >& blocks, const BlockedOptions& options) {
88 const auto nblocks = blocks.size();
89 auto block_weights = sanisizer::create<std::vector<Distance_> >(nblocks);
90 scran_blocks::compute_weights(
91 sanisizer::cast<std::size_t>(nblocks),
92 [&](std::size_t b) -> Index_ { return blocks[b].first; },
93 options.block_weight_policy,
94 options.variable_block_weight_parameters,
95 [&](std::size_t b, Distance_ w) -> void { block_weights[b] = w; }
96 );
97
98 const auto total_weight = [&]{
99 quickstats::PairwiseSumWorkspace<Distance_> pswrk;
100 quickstats::PairwiseSumOptions psopt;
101 return quickstats::pairwise_sum(block_weights.size(), block_weights.data(), pswrk, psopt);
102 }();
103
104 auto outputs = sanisizer::create<std::vector<std::pair<Distance_, Distance_> > >(nblocks);
105 knncolle::parallelize(options.num_threads, nblocks, [&](const int, I<decltype(nblocks)> start, I<decltype(nblocks)> length) -> void {
106 for (I<decltype(nblocks)> b = start, bend = start + length; b < bend; ++b) {
107 const auto curweight = block_weights[b];
108 const auto curdist = compute_distance(blocks[b].first, blocks[b].second);
109 outputs[b].first = curdist.first * curweight;
110 outputs[b].second = curdist.second * curweight;
111 }
112 });
113
114 std::pair<Distance_, Distance_> output{};
115 for (I<decltype(nblocks)> b = 0; b < nblocks; ++b) {
116 output.first += outputs[b].first;
117 output.second += outputs[b].second;
118 }
119
120 if (total_weight) {
121 output.first /= total_weight;
122 output.second /= total_weight;
123 }
124
125 return output;
126}

147template<typename Index_, typename Input_, typename Distance_>
148std::pair<Distance_, Distance_> compute_distance_blocked(
149 const std::vector<std::shared_ptr<const knncolle::Prebuilt<Index_, Input_, Distance_> > >& prebuilts,
150 Distance_* const buffer,
151 const BlockedOptions& options
152) {
153 const auto nblocks = prebuilts.size();
154 std::size_t accumulated = 0;
155 std::vector<std::pair<Index_, Distance_*> > blocks;
156 blocks.reserve(nblocks);
157
158 for (I<decltype(nblocks)> b = 0; b < nblocks; ++b) {
159 const auto nobs = prebuilts[b]->num_observations();
160 const auto capped_k = knncolle::cap_k(options.num_neighbors, nobs);
161
162 knncolle::parallelize(options.num_threads, nobs, [&](const int, const Index_ start, const Index_ length) -> void {
163 const auto searcher = prebuilts[b]->initialize();
164 std::vector<Distance_> cur_distances;
165 for (Index_ i = start, end = start + length; i < end; ++i) {
166 searcher->search(i, capped_k, NULL, &cur_distances);
167 if (cur_distances.size()) {
168 buffer[accumulated + i] = cur_distances.back();
169 } else {
170 buffer[accumulated + i] = 0; // i.e., only distance is that to itself.
171 }
172 }
173 });
174
175 blocks.emplace_back(nobs, buffer + accumulated);
176 accumulated += nobs;
177 }
178
179 return compute_distance_blocked(blocks, options);
180}

206template<typename Index_, typename Input_, typename Block_, typename Distance_, class Matrix_ = knncolle::Matrix<Index_, Input_> >
207std::pair<Distance_, Distance_> compute_distance_blocked(
208 const std::size_t num_dim,
209 const Index_ num_cells,
210 const Input_* const data,
211 const Block_* const blocks,
212 const std::size_t num_blocks,
213 const knncolle::Builder<Index_, Input_, Distance_, Matrix_>& builder,
214 Distance_* const buffer,
215 const BlockedOptions& options
216) {
217 // Avoiding allocation of a temporary buffer if we're already dealing with contiguous blocks.
218 auto block_details = sanisizer::create<std::vector<std::pair<Index_, Index_> > >(num_blocks);
219 Index_ non_contiguous = 0;
220 for (Index_ c = 0; c < num_cells; ++c) {
221 auto& curblock = block_details[blocks[c]];
222 if (curblock.second == 0) {
223 curblock.first = c;
224 curblock.second = 1;
225 } else {
226 non_contiguous += (c != curblock.first + curblock.second);
227 ++curblock.second;
228 }
229 }
230
231 const Input_* dataptr = data;
232 std::optional<std::vector<Input_> > tmp_data;
233 if (non_contiguous) {
234 // Otherwise, we reorganize the data so that observations from the same batch are in a single block.
235 Index_ accumulated = 0;
236 auto offsets = sanisizer::create<std::vector<Index_> >(num_blocks);
237 for (std::size_t b = 0; b < num_blocks; ++b) {
238 offsets[b] = accumulated;
239 block_details[b].first = accumulated;
240 accumulated += block_details[b].second; // this won't overflow as we already know that num_cells fits in an Index_.
241 }
242
243 tmp_data.emplace(sanisizer::product<typename std::vector<Input_>::size_type>(num_dim, num_cells));
244 for (Index_ c = 0; c < num_cells; ++c) {
245 auto& off = offsets[blocks[c]];
246 std::copy_n(
247 data + sanisizer::product_unsafe<std::size_t>(c, num_dim),
248 num_dim,
249 tmp_data->data() + sanisizer::product_unsafe<std::size_t>(off, num_dim)
250 );
251 ++off;
252 }
253
254 dataptr = tmp_data->data();
255 }
256
257 auto prebuilts = sanisizer::create<std::vector<std::shared_ptr<const knncolle::Prebuilt<Index_, Input_, Distance_> > > >(num_blocks);
258 knncolle::parallelize(options.num_threads, num_blocks, [&](const int, const std::size_t start, const std::size_t length) -> void {
259 for (std::size_t b = start, end = start + length; b < end; ++b) {
260 const auto sofar = block_details[b].first;
261 const auto cursize = block_details[b].second;
262 prebuilts[b] = builder.build_shared(knncolle::SimpleMatrix(num_dim, cursize, dataptr + sanisizer::product_unsafe<std::size_t>(sofar, num_dim)));
263 }
264 });
265
266 return compute_distance_blocked(prebuilts, buffer, options);
267}
