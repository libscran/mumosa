138std::pair<Distance_, Distance_> compute_distance_blocked(
139 const std::vector<std::shared_ptr<const knncolle::Prebuilt<Index_, Input_, Distance_> > >& prebuilts,
140 BlockedWorkspace<Distance_>& workspace,
141 const BlockedOptions& options
142) {
143 Options simple_opt;
144 simple_opt.num_neighbors = options.num_neighbors;
145 simple_opt.num_threads = options.num_threads;
146
147 std::pair<Distance_, Distance_> output(0, 0);
148
149 const auto nblocks = prebuilts.size();
150 for (I<decltype(nblocks)> b = 0; b < nblocks; ++b) {
151 const auto curweight = workspace.weights[b];
152 const auto& pbptr = prebuilts[b];
153 if (curweight && pbptr && pbptr->num_observations()) {
154 const auto curdist = compute_distance(*pbptr, workspace.distance_buffer.data(), simple_opt);
155 output.first += curdist.first * curweight;
156 output.second += curdist.second * curweight;
157 }
158 }
159
160 if (workspace.total_weight) {
161 output.first /= workspace.total_weight;
162 output.second /= workspace.total_weight;
163 }
164
165 return output;

240std::pair<Distance_, Distance_> compute_distance_blocked(
241 const std::size_t num_dim,
242 const Index_ num_cells,
243 const Input_* const data,
244 const Block_* const block,
245 const knncolle::Builder<Index_, Input_, Distance_, Matrix_>& builder,
246 const BlockedOptions& options
247) {
248 Block_ num_blocks = 0;
249 if (num_cells) {
250 num_blocks = sanisizer::sum<Block_>(1, *std::max_element(block, block + num_cells));
251 }
252
253 auto block_ends = sanisizer::create<std::vector<Index_> >(num_blocks);
254 auto block_non_contig = sanisizer::create<std::vector<char> >(num_blocks);
255 auto block_sizes = sanisizer::create<std::vector<Index_> >(num_blocks);
256
257 for (Index_ c = 0; c < num_cells; ++c) {
258 const auto curb = block[c];
259 block_sizes[curb] += 1;
260
261 auto& nc = block_non_contig[curb];
262 if (!nc) {
263 auto& be = block_ends[curb];
264 if (be == 0) {
265 be = c + 1;
266 } else if (be == c) {
267 ++be;
268 } else {
269 nc = true;
270 }
271 }
272 }
273
274 auto prebuilts = sanisizer::create<std::vector<std::shared_ptr<const knncolle::Prebuilt<Index_, Input_, Distance_> > > >(num_blocks);
275 Index_ non_contig_offset = 0;
276
277 for (Block_ b = 0; b < num_blocks; ++b) {
278 const auto length = block_sizes[b];
279 if (block_non_contig[b]) {
280 block_ends[b] = non_contig_offset; // repurposing block_ends to store starting offsets for each non-contiguous block.
281 non_contig_offset += length;
282 } else if (length) {
283 const auto start = block_ends[b] - length;
284 const auto ptr = data + sanisizer::product_unsafe<std::size_t>(start, num_dim);
285 prebuilts[b] = builder.build_shared(knncolle::SimpleMatrix(num_dim, length, ptr));
286 }
287 }
288
289 // Handling the non-contiguous blocks.
290 if (non_contig_offset > 0) {
291 std::vector<Input_> tmp_buffer(sanisizer::product<typename std::vector<Input_>::size_type>(non_contig_offset, num_dim));
292
293 for (Index_ c = 0; c < num_cells; ++c) {
294 const auto curb = block[c];
295 if (block_non_contig[curb]) {
296 auto& curoff = block_ends[curb];
297 std::copy_n(
298 data + sanisizer::product_unsafe<std::size_t>(c, num_dim),
299 num_dim,
300 tmp_buffer.data() + sanisizer::product_unsafe<std::size_t>(curoff, num_dim)
301 );
302 ++curoff;
303 }
304 }
305
306 non_contig_offset = 0;
307 for (Block_ b = 0; b < num_blocks; ++b) {
308 if (block_non_contig[b]) {
309 const auto length = block_sizes[b];
310 const auto ptr = tmp_buffer.data() + sanisizer::product_unsafe<std::size_t>(non_contig_offset, num_dim);
311 prebuilts[b] = builder.build_shared(knncolle::SimpleMatrix(num_dim, length, ptr));
312 non_contig_offset += length;
313 }
314 }
315 }
316
317 auto workspace = create_workspace<Distance_>(block_sizes, options);
318 return compute_distance_blocked(prebuilts, workspace, options);
319}
