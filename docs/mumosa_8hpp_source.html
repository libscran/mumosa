56template<typename Index_, typename Distance_>
57std::pair<Distance_, Distance_> compute_distance(Index_ num_cells, Distance_* distances) {
58 Distance_ med = tatami_stats::medians::direct(distances, num_cells, /* skip_nan = */ false);
59 Distance_ rmsd = 0;
60 for (Index_ i = 0; i < num_cells; ++i) {
61 auto d = distances[i];
62 rmsd += d * d;
63 }
64 rmsd = std::sqrt(rmsd);
65 return std::make_pair(med, rmsd);
66}

148template<typename Distance_>
149Distance_ compute_scale(const std::pair<Distance_, Distance_>& ref, const std::pair<Distance_, Distance_>& target) {
150 if (target.first == 0 || ref.first == 0) {
151 if (target.second == 0) {
152 return std::numeric_limits<Distance_>::infinity();
153 } else if (ref.second == 0) {
154 return 0;
155 } else {
156 return ref.second / target.second;
157 }
158 } else {
159 return ref.first / target.first;
160 }

176template<typename Distance_>
177std::vector<Distance_> compute_scale(const std::vector<std::pair<Distance_, Distance_> >& distances) {
178 std::vector<Distance_> output(distances.size());
179
180 // Use the first entry with a non-zero RMSD as the reference.
181 bool found_ref = false;
182 auto ndist = distances.size();
183 decltype(ndist) ref = 0;
184 for (decltype(ndist) e = 0; e < ndist; ++e) {
185 if (distances[e].second) {
186 found_ref = true;
187 ref = e;
188 break;
189 }
190 }
191
192 // If all of them have a zero RMSD, then all scalings are zero, because it doesn't matter.
193 if (found_ref) {
194 const auto& dref = distances[ref];
195 for (decltype(ndist) e = 0; e < ndist; ++e) {
196 output[e] = (e == ref ? 1 : compute_scale(dref, distances[e]));
197 }
198 }
199
200 return output;
201}

227 if (embeddings.size() != nembed || scaling.size() != nembed) {
228 throw std::runtime_error("'num_dims', 'embeddings' and 'scale' should have the same length");
229 }
230
231 std::size_t ntotal = std::accumulate(num_dims.begin(), num_dims.end(), static_cast<std::size_t>(0));
232 std::size_t offset = 0;
233
234 for (decltype(nembed) e = 0; e < nembed; ++e) {
235 auto curdim = num_dims[e];
236 auto inptr = embeddings[e];
237 auto s = scaling[e];
238
239 // We use offsets to avoid forming invalid pointers with strided pointers.
240 std::size_t in_position = 0;
241 std::size_t out_position = offset;
242
243 if (std::isinf(s)) {
244 // If the scaling factor is infinite, it implies that the current
245 // embedding is all-zero, so we just fill with zeros, and move on.
246 for (Index_ c = 0; c < num_cells; ++c, in_position += curdim, out_position += ntotal) {
247 std::fill_n(output + out_position, curdim, 0);
248 }
249 } else {
250 for (Index_ c = 0; c < num_cells; ++c, in_position += curdim, out_position += ntotal) {
251 for (std::size_t d = 0; d < curdim; ++d) {
252 output[out_position + d] = inptr[in_position + d] * s;
253 }
254 }
255 }

257 offset += curdim;
258 }
