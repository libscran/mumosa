51template<typename Index_, typename Distance_>
52std::pair<Distance_, Distance_> compute_distance(const Index_ num_cells, Distance_* const distances) {
53 const Distance_ med = tatami_stats::medians::direct(distances, num_cells, /* skip_nan = */ false);
54 Distance_ rmsd = 0;
55 for (Index_ i = 0; i < num_cells; ++i) {
56 const auto d = distances[i];
57 rmsd += d * d;
58 }
59 rmsd = std::sqrt(rmsd);
60 return std::make_pair(med, rmsd);
61}
